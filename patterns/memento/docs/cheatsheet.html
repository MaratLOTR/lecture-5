Memento — Снимок (самостоятельно)

SceneSettings хранит яркость, целевую температуру и список строк с именами устройств. save() создаёт неизменяемый SceneSnapshot, restore() восстанавливает настройки. Originator — SceneSettings, Memento — SceneSnapshot, Caretaker — History (задание). История должна хранить снимки и управлять ими через save()/restore(), не читая поля снимка.

Список устройств превращается в tuple: последующее изменение исходного списка не изменяет снимок. restore() создаёт новый список, поэтому повторное восстановление тоже независимо. Для вложенных изменяемых объектов понадобился бы другой способ копирования. Поля с подчёркиванием закрыты по соглашению Python, а не защищены от чтения. restore() принимает SceneSnapshot, иначе TypeError; снимки совместимы между сценами. Проверку диапазонов яркости и температуры пример не выполняет: используются корректные настройки. Восстановление объекта не отменяет команды реальному оборудованию.

# Самостоятельное задание: история настроек

В scene.py реализуйте History.checkpoint(), undo(), redo().
checkpoint() вызывается ПЕРЕД изменением: добавляет scene.save() в _undo и очищает _redo.
undo(): если _undo пуст, верните False без изменения сцены. Иначе сохраните текущее
состояние в _redo, восстановите последний снимок _undo и верните True.
redo(): симметричная операция, переносящая текущее состояние в _undo.
Используйте только save()/restore(), не обращайтесь к полям SceneSnapshot.

Проверка: A, checkpoint, изменение в B, checkpoint, изменение в C; undo даёт B,
undo даёт A, redo даёт B. После checkpoint и изменения в D redo возвращает False.
Не меняйте тесты и рабочую реализацию снимка. Подготовьте объяснение копирования,
ограничения памяти и отличий от обратимых команд.

"""Memento: независимый снимок настроек сцены. History — задание."""

from dataclasses import dataclass


@dataclass(frozen=True)
class SceneSnapshot:
    _brightness: int
    _target_temperature: float
    _devices: tuple[str, ...]


class SceneSettings:
    def __init__(self, brightness=50, target_temperature=22, devices=None):
        self.brightness = brightness
        self.target_temperature = target_temperature
        self.devices = list(devices) if devices is not None else []

    def save(self):
        return SceneSnapshot(self.brightness, self.target_temperature, tuple(self.devices))

    def restore(self, snapshot):
        if not isinstance(snapshot, SceneSnapshot):
            raise TypeError("Ожидается SceneSnapshot")
        self.brightness = snapshot._brightness
        self.target_temperature = snapshot._target_temperature
        self.devices = list(snapshot._devices)

import unittest
from dataclasses import FrozenInstanceError
from ..scene import SceneSettings


class SnapshotTests(unittest.TestCase):
    def test_snapshot_is_independent_and_restores_all_fields(self):
        devices = ["lamp"]
        scene = SceneSettings(50, 22, devices)
        devices.append("external")
        self.assertEqual(scene.devices, ["lamp"])
        snapshot = scene.save()
        scene.brightness, scene.target_temperature = 10, 18
        scene.devices.append("projector")
        scene.restore(snapshot)
        self.assertEqual((scene.brightness, scene.target_temperature, scene.devices), (50, 22, ["lamp"]))
        scene.devices.clear()
        scene.restore(snapshot)
        self.assertEqual(scene.devices, ["lamp"])

    def test_snapshot_is_frozen(self):
        snapshot = SceneSettings().save()
        with self.assertRaises(FrozenInstanceError):
            snapshot._brightness = 10

    def test_invalid_snapshot_does_not_partially_restore(self):
        scene = SceneSettings()
        before = scene.save()
        with self.assertRaises(TypeError):
            scene.restore(object())
        self.assertEqual(scene.save(), before)
tests/test_extra.py
import unittest
from ..scene import SceneSettings
from ..memento_limited_history import LimitedHistory


class LimitedHistoryTests(unittest.TestCase):
    def test_limit_discards_oldest_version(self):
        scene = SceneSettings(0)
        history = LimitedHistory(scene, limit=2)
        for value in (1, 2, 3):
            history.checkpoint()
            scene.brightness = value
        self.assertTrue(history.undo())
        self.assertEqual(scene.brightness, 2)
        self.assertTrue(history.undo())
        self.assertEqual(scene.brightness, 1)
        self.assertFalse(history.undo())
        self.assertTrue(history.redo())
        self.assertEqual(scene.brightness, 2)

    def test_invalid_limits(self):
        for limit in (0, -1, 1.5, True):
            with self.assertRaises(ValueError):
                LimitedHistory(SceneSettings(), limit)

import unittest
from ..scene import SceneSettings, History


class HistoryTests(unittest.TestCase):
    def test_empty_history(self):
        history = History(SceneSettings())
        self.assertIs(history.undo(), False)
        self.assertIs(history.redo(), False)

    def test_undo_redo_and_new_branch(self):
        scene = SceneSettings(10, devices=["A"])
        history = History(scene)
        history.checkpoint()
        scene.brightness, scene.devices = 20, ["B"]
        history.checkpoint()
        scene.brightness, scene.devices = 30, ["C"]
        self.assertIs(history.undo(), True)
        self.assertEqual((scene.brightness, scene.devices), (20, ["B"]))
        self.assertTrue(history.undo())
        self.assertEqual(scene.brightness, 10)
        self.assertFalse(history.undo())
        self.assertTrue(history.redo())
        self.assertEqual(scene.brightness, 20)
        history.checkpoint()
        scene.brightness = 40
        self.assertFalse(history.redo())
        self.assertTrue(history.undo())
        self.assertEqual(scene.brightness, 20)
