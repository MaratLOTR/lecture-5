Observer — Наблюдатель

TemperatureSensor хранит подписчиков с методом update(value). Display печатает температуру, Log хранит её в values. Подписка не отправляет начальное значение. Повторная подписка не дублируется, повторная отписка допустима. set_value() сначала меняет value, затем синхронно уведомляет подписчиков. Одинаковое значение не создаёт событие. В демонстрации журнал получает [21, 22].

Роли: TemperatureSensor — Subject, update(value) — контракт Observer, Display и Log — ConcreteObserver. Здесь используется Push: издатель передаёт значение. Pull-вариант передал бы источник, из которого подписчик прочитал бы данные. Подписчик реализует метод по соглашению, отдельный Protocol не требуется.

Обход tuple(_observers) фиксирует состав текущей рассылки. Новая подписка или отписка внутри update() влияет на следующую рассылку. Исключение прерывает текущую рассылку, но значение датчика уже обновлено. Ссылки удерживаются до отписки. Порядок соответствует порядку подписки. Многопоточность, повторные попытки и вложенные рассылки не регулируются. Классы примеров не переопределяют __eq__: проверка повторной подписки сравнивает объекты.

"""Observer: синхронная подписка на изменения температуры."""

class TemperatureSensor:
    def __init__(self, value):
        self.value = value
        self._observers = []

    def subscribe(self, observer):
        if observer not in self._observers:
            self._observers.append(observer)

    def unsubscribe(self, observer):
        if observer in self._observers:
            self._observers.remove(observer)

    # Продолжение TemperatureSensor
    def set_value(self, value):
        if value == self.value:
            return
        self.value = value
        for observer in tuple(self._observers):
            observer.update(value)

class Display:
    def update(self, value):
        print(f"Temperature: {value}")

class Log:
    def __init__(self):
        self.values = []

    def update(self, value):
        self.values.append(value)

import unittest
from ..sensor import TemperatureSensor, Log


class ObserverTests(unittest.TestCase):
    def test_subscription_repeat_value_and_unsubscribe(self):
        sensor, log = TemperatureSensor(20), Log()
        sensor.subscribe(log)
        sensor.subscribe(log)
        self.assertEqual(log.values, [])
        for value in (21, 21, 22):
            sensor.set_value(value)
        self.assertEqual(log.values, [21, 22])
        sensor.unsubscribe(log)
        sensor.unsubscribe(log)
        sensor.set_value(23)
        self.assertEqual(log.values, [21, 22])

    def test_mutation_of_subscriptions_affects_next_delivery(self):
        sensor, later, added = TemperatureSensor(20), Log(), Log()
        class Editor:
            def update(self, value):
                sensor.unsubscribe(later)
                sensor.subscribe(added)
        sensor.subscribe(Editor())
        sensor.subscribe(later)
        sensor.set_value(21)
        self.assertEqual(later.values, [21])
        self.assertEqual(added.values, [])
        sensor.set_value(22)
        self.assertEqual(later.values, [21])
        self.assertEqual(added.values, [22])

    def test_error_stops_delivery_but_value_is_updated(self):
        sensor, log = TemperatureSensor(20), Log()
        class Broken:
            def update(self, value):
                raise RuntimeError("subscriber error")
        sensor.subscribe(Broken())
        sensor.subscribe(log)
        with self.assertRaises(RuntimeError):
            sensor.set_value(21)
        self.assertEqual(sensor.value, 21)
        self.assertEqual(log.values, [])
