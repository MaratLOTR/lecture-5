State — Состояние

SecuritySystem хранит текущее состояние и делегирует ему handle(context, event). Disarmed, Armed и Alarm задают разные реакции на arm, motion и disarm. В Disarmed движение игнорируется, в Armed вызывает Alarm. В Alarm сирена работает до disarm. transition_to() синхронизирует состояние и сирену. Повторные команды сохраняют состояние. Неизвестное событие вызывает ValueError без изменений.

Роли: SecuritySystem — Context, SecurityState — базовый контракт (метод вызывает NotImplementedError), Disarmed/Armed/Alarm — ConcreteState. Правила переходов здесь находятся в самих состояниях. Новые экземпляры создаются при переходе. Прямое присваивание state обходит синхронизацию сирены: используйте transition_to().

Turnstile — простой конечный автомат из лекции: locked/unlocked и события coin/passage. coin открывает заблокированный турникет, passage после оплаты пропускает и закрывает. Попытка без оплаты блокируется, повторная монета при открытом проходе возвращается. Результаты handle(): open, blocked, pass_and_close, return_coin. Это отдельная реализация через условия: машина состояний — модель, State — способ организовать её код. Наличие автомата не обязывает создавать классы состояний.

security.py
"""State: охрана и простой конечный автомат турникета."""

class SecuritySystem:
    def __init__(self):
        self.state = Disarmed()
        self.siren = False

    def transition_to(self, state):
        self.state = state
        self.siren = isinstance(state, Alarm)

    def handle(self, event):
        if event not in {"arm", "motion", "disarm"}:
            raise ValueError(event)
        self.state.handle(self, event)

class SecurityState:
    def handle(self, context, event):
        raise NotImplementedError

class Disarmed(SecurityState):
    def handle(self, context, event):
        if event == "arm":
            context.transition_to(Armed())
        # motion и disarm: без изменений

class Armed(SecurityState):
    def handle(self, context, event):
        if event == "motion":
            context.transition_to(Alarm())
        elif event == "disarm":
            context.transition_to(Disarmed())

class Alarm(SecurityState):
    def handle(self, context, event):
        if event == "disarm":
            context.transition_to(Disarmed())

class Turnstile:
    """Правила из лекции. Событие passage означает попытку пройти."""

    def __init__(self):
        self.state = "locked"

    def handle(self, event):
        if event not in {"coin", "passage"}:
            raise ValueError(event)
        if self.state == "locked":
            if event == "coin":
                self.state = "unlocked"
                return "open"
            return "blocked"
        if event == "passage":
            self.state = "locked"
            return "pass_and_close"
        return "return_coin"

tests/test_example.py
import unittest
from ..security import SecuritySystem, Disarmed, Armed, Alarm, Turnstile
from .. import security as m


class SecurityTests(unittest.TestCase):
    def test_all_supported_transitions_and_siren(self):
        table = {
            Disarmed: {"arm": getattr(m, "Arming", Armed), "motion": Disarmed, "disarm": Disarmed},
            Armed: {"arm": Armed, "motion": Alarm, "disarm": Disarmed},
            Alarm: {"arm": Alarm, "motion": Alarm, "disarm": Disarmed},
        }
        for initial, transitions in table.items():
            for event, expected in transitions.items():
                with self.subTest(initial=initial.__name__, event=event):
                    system = SecuritySystem()
                    system.transition_to(initial())
                    system.handle(event)
                    self.assertIsInstance(system.state, expected)
                    self.assertEqual(system.siren, expected is Alarm)

    def test_unknown_event_preserves_state(self):
        system = SecuritySystem()
        state = system.state
        with self.assertRaises(ValueError):
            system.handle("unknown")
        self.assertIs(system.state, state)
        self.assertFalse(system.siren)


class TurnstileTests(unittest.TestCase):
    def test_full_transition_table(self):
        for initial, event, result, final in (
            ("locked", "coin", "open", "unlocked"),
            ("locked", "passage", "blocked", "locked"),
            ("unlocked", "coin", "return_coin", "unlocked"),
            ("unlocked", "passage", "pass_and_close", "locked"),
        ):
            with self.subTest(initial=initial, event=event):
                turnstile = Turnstile()
                if initial == "unlocked":
                    turnstile.handle("coin")
                self.assertEqual(turnstile.handle(event), result)
                self.assertEqual(turnstile.state, final)

    def test_one_payment_cannot_buy_two_passages(self):
        turnstile = Turnstile()
        for event, result in (("coin", "open"), ("passage", "pass_and_close"), ("passage", "blocked")):
            self.assertEqual(turnstile.handle(event), result)
        with self.assertRaises(ValueError):
            turnstile.handle("unknown")
        self.assertEqual(turnstile.state, "locked")

import unittest
from .. import security as m


class ArmingTests(unittest.TestCase):
    def setUp(self):
        self.assertIsNotNone(getattr(m, "Arming", None), "ЗАДАНИЕ: добавьте Arming и timeout, см. TASK.md")

    def test_waiting_then_alarm(self):
        system = m.SecuritySystem()
        system.handle("arm")
        for event in ("arm", "motion"):
            system.handle(event)
            self.assertIsInstance(system.state, m.Arming)
            self.assertFalse(system.siren)
        system.handle("timeout")
        self.assertIsInstance(system.state, m.Armed)
        system.handle("motion")
        self.assertIsInstance(system.state, m.Alarm)
        self.assertTrue(system.siren)

    def test_cancellation_and_timeout_elsewhere(self):
        system = m.SecuritySystem()
        system.handle("arm")
        system.handle("disarm")
        self.assertIsInstance(system.state, m.Disarmed)
        for state in (m.Disarmed(), m.Armed(), m.Alarm()):
            system.transition_to(state)
            system.handle("timeout")
            self.assertIs(system.state, state)
            self.assertEqual(system.siren, isinstance(state, m.Alarm))
